const LATEST_HANDBOOK = /player.s handbook \(2024\)/i;
/** The free part of that book, which every install has. */
const FREE_2024 = /^SRD 5\.2/i;

/**
 * What a guided step offers. Few (the default): the latest Player's Handbook where its content is loaded, else the free
 * 2024 rules. All: every official book's, one entry per name (the newest). Playtest, third-party and homebrew
 * content is never offered, and what the content hides (a sidekick class, an NPC species) stays hidden.
 */
export function guideOptions(reg: ARegistry, books: BookIndex, kind: GuideKind, all: boolean): AElement[] {
  const shown = reg.ofType(kind).filter((e) => !e.compendiumHidden);
  const official = shown.filter((e) => {
    const k = bookKind(books.info(e.source), e.source);
    return k === "official" || k === "setting" || FREE_2024.test(e.source);
  });
  const byName = (list: AElement[]) => [...list].sort((a, b) => a.name.localeCompare(b.name));
  const handbook = official.filter((e) => LATEST_HANDBOOK.test(e.source));
  const few = handbook.length ? handbook : official.filter((e) => FREE_2024.test(e.source));
  if (!all) return byName(few.length ? few : official);
  // One entry per name: the latest handbook's, else the newest book's.
  const best = new Map<string, AElement>();
  const rank = (e: AElement) => (LATEST_HANDBOOK.test(e.source) ? 3 : FREE_2024.test(e.source) ? 2 : 1);
  const year = (e: AElement) => books.info(e.source).release ?? 0;
  for (const e of official) {
    const k = e.name.toLowerCase().replace(/\s*\((?:20\d\d|legacy)\)\s*$/, "");
    const have = best.get(k);
    if (!have || rank(e) > rank(have) || (rank(e) === rank(have) && year(e) > year(have))) best.set(k, e);
  }
  return byName([...best.values()]);
}

/** The line shown under an option: the guide's own for the handbook's options, else the option's first sentence. */
export function guideLine(kind: GuideKind, el: AElement): string {
  const own = GUIDE_LINES[kind][el.name];
  if (own) return own;
  const text = (el.description ?? "")
    .replace(/<[^>]+>/g, " ")
    .replace(/\s+/g, " ")
    .trim();
  const first = /^(.{20,160}?[.!?])(?:\s|$)/.exec(text)?.[1] ?? "";
  return first;
}

const S = "strength";
const D = "dexterity";
const C = "constitution";
const I = "intelligence";
const W = "wisdom";
const H = "charisma";
/** The standard array for each class, highest score first. */
const SCORES: Record<string, AbilityId[]> = {
  Barbarian: [S, C, D, W, H, I],
  Bard: [H, D, C, W, I, S],
  Cleric: [W, C, S, H, D, I],
  Druid: [W, C, D, I, H, S],
  Fighter: [S, C, D, W, H, I],
  Monk: [D, W, C, S, I, H],
  Paladin: [S, H, C, W, D, I],
  Ranger: [D, W, C, S, I, H],
  Rogue: [D, C, H, W, I, S],
  Sorcerer: [H, C, D, W, I, S],
  Warlock: [H, C, D, W, I, S],
  Wizard: [I, C, D, W, H, S],
};
export const GUIDE_CLASSES = Object.keys(SCORES);

const ARRAY = [15, 14, 13, 12, 10, 8];
/** The standard array placed for a class: its best score where the class needs it most. */
export function recommendedScores(className: string): Record<AbilityId, number> | undefined {
  const order = SCORES[className];
  return order ? (Object.fromEntries(order.map((a, i) => [a, ARRAY[i]])) as Record<AbilityId, number>) : undefined;
}
/** The score a class wants highest, for the guide's sentence about it. */
export const mainScore = (className: string | undefined): AbilityId | undefined => (className ? SCORES[className]?.[0] : undefined);
